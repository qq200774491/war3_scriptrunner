# 发行构建使用完整链接，避免依赖上次构建遗留的 .ilk 增量数据库。
add_link_options("$<$<CONFIG:Release>:/INCREMENTAL:NO>")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/compatibility/engine-v1.json" ENGINE_PROFILE)
string(JSON VERIFIED_SHA256 GET "${ENGINE_PROFILE}" sha256)
string(JSON CREATE_EXPORT GET "${ENGINE_PROFILE}" create_export)
string(JSON DESTROY_EXPORT GET "${ENGINE_PROFILE}" destroy_export)
foreach(NAME load tick event)
  string(TOUPPER "${NAME}" KEY)
  string(JSON ${KEY}_SLOT GET "${ENGINE_PROFILE}" runner_slots ${NAME})
endforeach()
add_library(detours STATIC third_party/detours/src/detours.cpp third_party/detours/src/modules.cpp

target_include_directories(detours PUBLIC third_party/detours/src)
target_compile_definitions(detours PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
set_property(TARGET detours PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

target_compile_options(zz_shared_quicktest PRIVATE /W4 /WX /EHsc /utf-8)
